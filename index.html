// Load data from LocalStorage on startup, or start with empty array let inventory = JSON.parse(localStorage.getItem('macroStock')) || []; // Run this once to show existing data when page loads window.onload = () => { displayStock(inventory); }; function updateInventory() { const nameInput = document.querySelector('input[placeholder*="Alpine"]'); const qtyInput = document.querySelector('input[placeholder="0"]'); const priceInput = document.querySelector('input[placeholder="0.00"]'); const name = nameInput.value.trim(); const stock = qtyInput.value.trim(); const price = priceInput.value.trim(); if (name === "" || stock === "" || price === "") { alert("Please fill in all fields!"); return; } // Create product object const product = { name, stock, price }; // Add to our list inventory.push(product); // Save to LocalStorage (Permanent storage) localStorage.setItem('macroStock', JSON.stringify(inventory)); // Refresh the table and clear inputs displayStock(inventory); nameInput.value = ""; qtyInput.value = ""; priceInput.value = ""; } function searchStock() { const query = document.getElementById('searchInput').value.toLowerCase(); // Filter the inventory based on name const filteredData = inventory.filter(item => item.name.toLowerCase().includes(query) ); displayStock(filteredData); } function displayStock(data) { const tableBody = document.querySelector('tbody'); if (!tableBody) return; tableBody.innerHTML = ""; data.forEach((item, index) => { const row = ` ${item.name} ${item.stock} R ${item.price} `; tableBody.innerHTML += row; }); }
